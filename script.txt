function Write-Step {
    <# In tien trinh buoc hien tai #>
    param(
        [string]$Message,
        [string]$Status = "INFO"   # INFO, OK, SKIP, FAIL
    )
    $script:currentStep++
    $prefix = "[Buoc $($script:currentStep)/$($script:totalSteps)]"
    switch ($Status) {
        "OK"   { Write-Host "$prefix [OK]   $Message" -ForegroundColor Green }
        "SKIP" { Write-Host "$prefix [SKIP] $Message" -ForegroundColor Yellow }
        "FAIL" { Write-Host "$prefix [FAIL] $Message" -ForegroundColor Red }
        default { Write-Host "$prefix [....] $Message" -ForegroundColor Cyan }
    }
}

function Test-CommandExists {
    <# Kiem tra mot lenh co ton tai trong PATH khong #>
    param([string]$Command)
    $null -ne (Get-Command $Command -ErrorAction SilentlyContinue)
}

function Test-AppInstalledWinget {
    <# Kiem tra app da cai qua winget chua (theo ID) #>
    param([string]$WingetId)
    if (-not (Test-CommandExists "winget")) { return $false }
    $listOutput = winget list --id $WingetId --accept-source-agreements 2>$null
    # winget list tra ve bang co chua ID neu da cai
    return ($listOutput | Out-String) -match [regex]::Escape($WingetId)
}

function Test-AppInstalledChoco {
    <# Kiem tra app da cai qua Chocolatey chua #>
    param([string]$ChocoId)
    if (-not (Test-CommandExists "choco")) { return $false }
    $listOutput = choco list --local-only --exact $ChocoId 2>$null
    return ($listOutput | Out-String) -match $ChocoId
}

function Install-WithWinget {
    <# Cai app bang winget, tra ve $true neu thanh cong #>
    param([string]$WingetId)
    if (-not (Test-CommandExists "winget")) { return $false }
    try {
        $proc = Start-Process -FilePath "winget" `
            -ArgumentList "install --id $WingetId --accept-package-agreements --accept-source-agreements --silent" `
            -Wait -PassThru -NoNewWindow
        return ($proc.ExitCode -eq 0)
    } catch {
        Write-Host "    Winget install loi: $_" -ForegroundColor Red
        return $false
    }
}

function Install-WithChoco {
    <# Cai app bang Chocolatey, tra ve $true neu thanh cong #>
    param([string]$ChocoId)
    if (-not (Test-CommandExists "choco")) { return $false }
    try {
        $proc = Start-Process -FilePath "choco" `
            -ArgumentList "install $ChocoId -y --no-progress" `
            -Wait -PassThru -NoNewWindow
        return ($proc.ExitCode -eq 0)
    } catch {
        Write-Host "    Choco install loi: $_" -ForegroundColor Red
        return $false
    }
}

function Search-PackageWinget {
    <# Tim app tren winget, tra ve ID neu tim thay ket qua khop chinh xac #>
    param([string]$AppName)
    if (-not (Test-CommandExists "winget")) { return $null }
    try {
        $searchOutput = winget search --name $AppName --accept-source-agreements 2>$null
        $outputStr = $searchOutput | Out-String
        # Tim dong co chua ten app (khop chinh xac - ten bat dau dong)
        $lines = $outputStr -split "`n" | Where-Object { $_.Trim() -ne "" }
        # Bo qua header (2 dong dau), tim dong co ten khop chinh xac
        foreach ($line in $lines) {
            # Winget output co dang: Name   Id   Version   Source
            # Kiem tra ten khop chinh xac (case-insensitive)
            if ($line -match "^\s*$([regex]::Escape($AppName))\s+(\S+)\s+" ) {
                $foundId = $Matches[1]
                Write-Host "    Winget search tim thay: $foundId" -ForegroundColor Gray
                return $foundId
            }
        }
        return $null
    } catch {
        return $null
    }
}

function Search-PackageChoco {
    <# Tim app tren Chocolatey, tra ve ID neu tim thay ket qua khop chinh xac #>
    param([string]$AppName)
    if (-not (Test-CommandExists "choco")) { return $null }
    try {
        $searchOutput = choco search $AppName --exact 2>$null
        $outputStr = $searchOutput | Out-String
        # Choco search --exact tra ve: "appname 1.2.3 [Approved]"
        $lines = $outputStr -split "`n" | Where-Object { $_.Trim() -ne "" }
        foreach ($line in $lines) {
            if ($line -match "^($([regex]::Escape($AppName)))\s+[\d]") {
                $foundId = $Matches[1]
                Write-Host "    Choco search tim thay: $foundId" -ForegroundColor Gray
                return $foundId
            }
        }
        return $null
    } catch {
        return $null
    }
}

function Initialize-PackageManagers {
    <# Kiem tra va cai dat winget + Chocolatey #>

    # --- Kiem tra winget ---
    Write-Step "Kiem tra winget..." "INFO"
    if (Test-CommandExists "winget") {
        Write-Host "    winget da co san." -ForegroundColor Green
    } else {
        Write-Host "    [!] winget KHONG co. Ban can cai 'App Installer' tu Microsoft Store." -ForegroundColor Yellow
        Write-Host "    Mo Microsoft Store > Tim 'App Installer' > Cai dat." -ForegroundColor Yellow
        Write-Host "    Hoac chay: Add-AppxPackage -Path <duong-dan-msixbundle>" -ForegroundColor Yellow
        $null = $script:results.ManualNeeded.Add("winget (App Installer) - can cai tu Microsoft Store")
    }

    # --- Cai Chocolatey ---
    Write-Step "Kiem tra / Cai dat Chocolatey..." "INFO"
    if (Test-CommandExists "choco") {
        Write-Host "    Chocolatey da co san." -ForegroundColor Green
    } else {
        Write-Host "    Dang cai Chocolatey..." -ForegroundColor Cyan
        try {
            # Chi dat ExecutionPolicy trong scope Process (an toan)
            Set-ExecutionPolicy Bypass -Scope Process -Force
            [System.Net.ServicePointManager]::SecurityProtocol = `
                [System.Net.ServicePointManager]::SecurityProtocol -bor 3072
            $installScript = (New-Object System.Net.WebClient).DownloadString(
                'https://community.chocolatey.org/install.ps1'
            )
            Invoke-Expression $installScript
            # Lam moi PATH de nhan dien choco
            $env:Path = [System.Environment]::GetEnvironmentVariable("Path", "Machine") + ";" + `
                        [System.Environment]::GetEnvironmentVariable("Path", "User")
            if (Test-CommandExists "choco") {
                Write-Host "    Chocolatey cai thanh cong." -ForegroundColor Green
            } else {
                throw "Khong tim thay choco sau khi cai"
            }
        } catch {
            Write-Host "    Cai Chocolatey that bai: $_" -ForegroundColor Red
            $null = $script:results.Failed.Add("Chocolatey")
        }
    }
}

function Enable-WSL2ForDocker {
    <#
    Bat WSL va VirtualMachinePlatform cho Docker Desktop tren Win10 Home.
    Yeu cau build >= 19041.
    HOAN TAC: Disable-WindowsOptionalFeature -Online -FeatureName Microsoft-Windows-Subsystem-Linux
              Disable-WindowsOptionalFeature -Online -FeatureName VirtualMachinePlatform
    #>
    Write-Host "    [Docker Pre] Kiem tra yeu cau WSL2..." -ForegroundColor Cyan

    if ($osBuild -lt 19041) {
        Write-Host "    [!] Build Windows ($osBuild) < 19041. Khong the bat WSL2." -ForegroundColor Red
        throw "Windows build qua cu cho WSL2"
    }

    # Bat Microsoft-Windows-Subsystem-Linux
    $wslFeature = Get-WindowsOptionalFeature -Online -FeatureName Microsoft-Windows-Subsystem-Linux
    if ($wslFeature.State -ne "Enabled") {
        Write-Host "    Dang bat Windows Subsystem for Linux..." -ForegroundColor Cyan
        Enable-WindowsOptionalFeature -Online -FeatureName Microsoft-Windows-Subsystem-Linux -All -NoRestart | Out-Null
        $script:needReboot = $true
    } else {
        Write-Host "    WSL da duoc bat." -ForegroundColor Green
    }

    # Bat VirtualMachinePlatform
    $vmpFeature = Get-WindowsOptionalFeature -Online -FeatureName VirtualMachinePlatform
    if ($vmpFeature.State -ne "Enabled") {
        Write-Host "    Dang bat Virtual Machine Platform..." -ForegroundColor Cyan
        Enable-WindowsOptionalFeature -Online -FeatureName VirtualMachinePlatform -All -NoRestart | Out-Null
        $script:needReboot = $true
    } else {
        Write-Host "    VirtualMachinePlatform da duoc bat." -ForegroundColor Green
    }

    # Dat WSL default version = 2
    if (Test-CommandExists "wsl") {
        wsl --set-default-version 2 2>$null
        Write-Host "    WSL default version dat thanh 2." -ForegroundColor Green
    }

    # Canh bao
    Write-Host "" -ForegroundColor Yellow
    Write-Host "    ============================================================" -ForegroundColor Yellow
    Write-Host "    CANH BAO QUAN TRONG:" -ForegroundColor Yellow
    Write-Host "    1. Ban CAN bat Virtualization (VT-x / AMD-V) trong BIOS." -ForegroundColor Yellow
    Write-Host "    2. Sau khi script chay xong, CAN KHOI DONG LAI may" -ForegroundColor Yellow
    Write-Host "       de WSL2 hoat dong day du." -ForegroundColor Yellow
    Write-Host "    ============================================================" -ForegroundColor Yellow
    Write-Host ""
}

function Install-Applications {
    <# Cai tat ca app trong $appList #>
    foreach ($app in $appList) {
        $name = $app.Name
        Write-Step "Xu ly: $name" "INFO"

        try {
            # --- Truong hop can tim kiem (Search = $true) ---
            if ($app.Search -eq $true) {
                Write-Host "    Tim kiem '$name' tren winget va choco..." -ForegroundColor Cyan

                $wingetId = Search-PackageWinget -AppName $name
                $chocoId  = Search-PackageChoco -AppName $name

                if ($wingetId) {
                    $app.WingetId = $wingetId
                } elseif ($chocoId) {
                    $app.ChocoId = $chocoId
                } else {
                    Write-Host "    Khong tim thay goi chinh xac cho '$name'. Bo qua." -ForegroundColor Yellow
                    $null = $script:results.ManualNeeded.Add("$name - Khong tim thay tren winget/choco")
                    continue
                }
            }

            # --- Kiem tra da cai chua (idempotent) ---
            $alreadyInstalled = $false
            if ($app.WingetId -and (Test-AppInstalledWinget -WingetId $app.WingetId)) {
                $alreadyInstalled = $true
            }
            if (-not $alreadyInstalled -and $app.ChocoId -and (Test-AppInstalledChoco -ChocoId $app.ChocoId)) {
                $alreadyInstalled = $true
            }

            if ($alreadyInstalled) {
                Write-Step "$name - Da co san, bo qua." "SKIP"
                $null = $script:results.AlreadyHave.Add($name)
                continue
            }

            # --- Chay PreAction neu co (VD: bat WSL cho Docker) ---
            if ($app.PreAction) {
                Write-Host "    Chay PreAction cho $name..." -ForegroundColor Cyan
                & $app.PreAction
            }

            # --- Thu cai bang winget truoc ---
            $installed = $false
            if ($app.WingetId) {
                Write-Host "    Thu cai $name bang winget ($($app.WingetId))..." -ForegroundColor Cyan
                $installed = Install-WithWinget -WingetId $app.WingetId
            }

            # --- Fallback sang Chocolatey ---
            if (-not $installed -and $app.ChocoId) {
                Write-Host "    Winget that bai/khong co. Thu Chocolatey ($($app.ChocoId))..." -ForegroundColor Yellow
                $installed = Install-WithChoco -ChocoId $app.ChocoId
            }

            if ($installed) {
                Write-Step "$name - Cai thanh cong!" "OK"
                $null = $script:results.Installed.Add($name)
            } else {
                Write-Step "$name - THAT BAI ca winget lan choco." "FAIL"
                $null = $script:results.Failed.Add($name)
            }

            # Hien ghi chu neu co
            if ($app.Note) {
                Write-Host "    Ghi chu: $($app.Note)" -ForegroundColor Gray
            }

        } catch {
            Write-Step "$name - LOI: $_" "FAIL"
            $null = $script:results.Failed.Add("$name (Loi: $_)")
        }
    }
}

function Remove-Bloatware {
    <#
    Go cac app bloatware pho bien cua Win10.
    Chi go tu danh sach an toan, KHONG go Store, Calculator, Photos.
    HOAN TAC: Cai lai tu Microsoft Store.
    #>
    Write-Step "Go bloatware Win10..." "INFO"
    foreach ($appName in $bloatwareList) {
        try {
            $pkg = Get-AppxPackage -Name $appName -ErrorAction SilentlyContinue
            if ($pkg) {
                $pkg | Remove-AppxPackage -ErrorAction Stop
                Write-Host "    Da go: $appName" -ForegroundColor Green
            }
            # Cung go khoi provisioned (khong cai lai cho user moi)
            $provPkg = Get-AppxProvisionedPackage -Online -ErrorAction SilentlyContinue |
                       Where-Object { $_.PackageName -like "*$appName*" }
            if ($provPkg) {
                $provPkg | Remove-AppxProvisionedPackage -Online -ErrorAction SilentlyContinue | Out-Null
            }
        } catch {
            Write-Host "    Khong go duoc $appName : $_" -ForegroundColor Yellow
        }
    }
    Write-Host "    Hoan tat go bloatware." -ForegroundColor Green
}

function Set-TelemetryAndPrivacy {
    <#
    Giam telemetry va tat quang cao bang Registry.
    HOAN TAC cho tung key:
      - AllowTelemetry: dat lai ve 3 (Full)
      - AdvertisingInfo: dat Enabled ve 1
      - ContentDeliveryManager: dat cac gia tri ve 1
      - CloudContent: xoa key DisableWindowsConsumerFeatures
    #>
    Write-Step "Giam telemetry va tat quang cao..." "INFO"

    try {
        # --- Giam Telemetry xuong muc thap nhat co the tren Home (1 = Basic) ---
        # HOAN TAC: Set-ItemProperty -Path $telPath -Name "AllowTelemetry" -Value 3
        $telPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\DataCollection"
        if (-not (Test-Path $telPath)) { New-Item -Path $telPath -Force | Out-Null }
        Set-ItemProperty -Path $telPath -Name "AllowTelemetry" -Value 1 -Type DWord -Force
        Write-Host "    AllowTelemetry = 1 (Basic)." -ForegroundColor Green

        # --- Tat Advertising ID ---
        # HOAN TAC: Set-ItemProperty -Path $adIdPath -Name "Enabled" -Value 1
        $adIdPath = "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\AdvertisingInfo"
        if (-not (Test-Path $adIdPath)) { New-Item -Path $adIdPath -Force | Out-Null }
        Set-ItemProperty -Path $adIdPath -Name "Enabled" -Value 0 -Type DWord -Force
        Write-Host "    Advertising ID = tat." -ForegroundColor Green

        # --- Tat goi y ung dung (quang cao trong Start Menu) ---
        # HOAN TAC: Dat cac gia tri sau ve 1
        $cdmPath = "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\ContentDeliveryManager"
        if (Test-Path $cdmPath) {
            Set-ItemProperty -Path $cdmPath -Name "SystemPaneSuggestionsEnabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SilentInstalledAppsEnabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SoftLandingEnabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SubscribedContent-338388Enabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SubscribedContent-338389Enabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SubscribedContent-353694Enabled" -Value 0 -Type DWord -Force
            Set-ItemProperty -Path $cdmPath -Name "SubscribedContent-353696Enabled" -Value 0 -Type DWord -Force
            Write-Host "    Goi y ung dung / quang cao Start Menu = tat." -ForegroundColor Green
        }

        # --- Tat Windows Consumer Features (goi y cai app tu MS) ---
        # HOAN TAC: Remove-ItemProperty -Path $cloudPath -Name "DisableWindowsConsumerFeatures"
        $cloudPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\CloudContent"
        if (-not (Test-Path $cloudPath)) { New-Item -Path $cloudPath -Force | Out-Null }
        Set-ItemProperty -Path $cloudPath -Name "DisableWindowsConsumerFeatures" -Value 1 -Type DWord -Force
        Write-Host "    Windows Consumer Features = tat." -ForegroundColor Green

    } catch {
        Write-Host "    Loi khi chinh telemetry: $_" -ForegroundColor Red
    }
}

function Set-ExplorerSettings {
    <#
    Hien phan mo rong file va file an trong Explorer.
    HOAN TAC:
      - HideFileExt: dat lai ve 1
      - Hidden: dat lai ve 2
    #>
    Write-Step "Chinh Explorer: hien duoi file + file an..." "INFO"
    try {
        $explorerPath = "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Explorer\Advanced"

        # Hien phan mo rong file (0 = hien, 1 = an)
        Set-ItemProperty -Path $explorerPath -Name "HideFileExt" -Value 0 -Type DWord -Force
        Write-Host "    Hien duoi mo rong file = bat." -ForegroundColor Green

        # Hien file an (1 = hien, 2 = an)
        Set-ItemProperty -Path $explorerPath -Name "Hidden" -Value 1 -Type DWord -Force
        Write-Host "    Hien file an = bat." -ForegroundColor Green

    } catch {
        Write-Host "    Loi khi chinh Explorer: $_" -ForegroundColor Red
    }
}

function Set-HighPerformancePowerPlan {
    <#
    Dat power plan thanh High Performance.
    HOAN TAC: powercfg /setactive SCHEME_BALANCED
    #>
    Write-Step "Dat power plan High Performance..." "INFO"
    try {
        # GUID cua High Performance: 8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c
        $highPerfGuid = "8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c"

        # Kiem tra xem plan co ton tai khong, neu khong thi duplicate tu Balanced
        $plans = powercfg /list
        if (-not ($plans | Out-String).Contains($highPerfGuid)) {
            Write-Host "    High Performance plan khong co san, thu tao tu Balanced..." -ForegroundColor Yellow
            powercfg /duplicatescheme $highPerfGuid 2>$null
        }

        powercfg /setactive $highPerfGuid
        Write-Host "    Power plan = High Performance." -ForegroundColor Green
    } catch {
        Write-Host "    Loi khi dat power plan: $_" -ForegroundColor Red
    }

function Set-GitConfig {
    <# Cau hinh Git: user.name, user.email, autocrlf, defaultBranch, credential.helper #>
    Write-Step "Cau hinh Git..." "INFO"

    # Lam moi PATH de nhan dien git moi cai
    $env:Path = [System.Environment]::GetEnvironmentVariable("Path", "Machine") + ";" + `
                [System.Environment]::GetEnvironmentVariable("Path", "User")

    if (-not (Test-CommandExists "git")) {
        Write-Host "    Git chua duoc cai. Bo qua cau hinh." -ForegroundColor Yellow
        return
    }

    try {
        Write-Host ""
        Write-Host "    === CAU HINH GIT ===" -ForegroundColor Cyan
        $gitName  = Read-Host "    Nhap Git user.name (VD: Nguyen Van A)"
        $gitEmail = Read-Host "    Nhap Git user.email (VD: email@example.com)"

        if ($gitName) {
            git config --global user.name "$gitName"
            Write-Host "    user.name = $gitName" -ForegroundColor Green
        }
        if ($gitEmail) {
            git config --global user.email "$gitEmail"
            Write-Host "    user.email = $gitEmail" -ForegroundColor Green
        }

        git config --global core.autocrlf true
        git config --global init.defaultBranch main
        git config --global credential.helper manager
        Write-Host "    core.autocrlf = true" -ForegroundColor Green
        Write-Host "    init.defaultBranch = main" -ForegroundColor Green
        Write-Host "    credential.helper = manager" -ForegroundColor Green

    } catch {
        Write-Host "    Loi cau hinh Git: $_" -ForegroundColor Red
    }
}

function Set-JavaEnvironment {
    <#
    Do tim JDK, dat JAVA_HOME va them vao PATH.
    CHI chay neu phat hien JDK da cai tren may.
    HOAN TAC:
      [System.Environment]::SetEnvironmentVariable("JAVA_HOME", $null, "Machine")
      # Va xoa %JAVA_HOME%\bin khoi PATH thu cong
    #>
    Write-Step "Kiem tra va cau hinh Java (JAVA_HOME)..." "INFO"

    # Cac vi tri thuong gap cua JDK
    $jdkSearchPaths = @(
        "$env:ProgramFiles\Java"
        "$env:ProgramFiles\Eclipse Adoptium"
        "$env:ProgramFiles\AdoptOpenJDK"
        "$env:ProgramFiles\Zulu"
        "$env:ProgramFiles\Microsoft\jdk-*"
        "$env:ProgramFiles\Amazon Corretto"
        "${env:ProgramFiles(x86)}\Java"
    )

    $jdkHome = $null
    foreach ($searchPath in $jdkSearchPaths) {
        if (Test-Path $searchPath) {
            # Tim thu muc JDK (co chua bin\javac.exe)
            $jdkDirs = Get-ChildItem -Path $searchPath -Directory -ErrorAction SilentlyContinue |
                       Where-Object {
                           Test-Path (Join-Path $_.FullName "bin\javac.exe")
                       } |
                       Sort-Object Name -Descending  # Lay version moi nhat
            if ($jdkDirs) {
                $jdkHome = $jdkDirs[0].FullName
                break
            }
        }
    }

    if (-not $jdkHome) {
        Write-Host "    Khong tim thay JDK tren may. Bo qua." -ForegroundColor Yellow
        Write-Host "    Ban co the cai JDK sau roi chay lai script." -ForegroundColor Yellow
        $null = $script:results.ManualNeeded.Add("Java JDK - Chua cai, can cai thu cong")
        return
    }

    Write-Host "    Tim thay JDK tai: $jdkHome" -ForegroundColor Green

    try {
        # Dat JAVA_HOME (Machine scope)
        [System.Environment]::SetEnvironmentVariable("JAVA_HOME", $jdkHome, "Machine")
        $env:JAVA_HOME = $jdkHome
        Write-Host "    JAVA_HOME = $jdkHome" -ForegroundColor Green

        # Them %JAVA_HOME%\bin vao PATH (Machine) neu chua co
        $machinePath = [System.Environment]::GetEnvironmentVariable("Path", "Machine")
        $javaBinEntry = "%JAVA_HOME%\bin"
        $javaBinActual = "$jdkHome\bin"

        # Kiem tra xem da co trong PATH chua (ca dang bien lan dang tuyet doi)
        $pathEntries = $machinePath -split ";"
        $alreadyInPath = $false
        foreach ($entry in $pathEntries) {
            $expanded = [System.Environment]::ExpandEnvironmentVariables($entry)
            if ($expanded -eq $javaBinActual -or $entry -eq $javaBinEntry) {
                $alreadyInPath = $true
                break
            }
        }

        if (-not $alreadyInPath) {
            $newPath = $machinePath.TrimEnd(";") + ";$javaBinEntry"
            [System.Environment]::SetEnvironmentVariable("Path", $newPath, "Machine")
            Write-Host "    Da them %JAVA_HOME%\bin vao PATH (Machine)." -ForegroundColor Green
        } else {
            Write-Host "    %JAVA_HOME%\bin da co trong PATH." -ForegroundColor Green
        }

        # Cap nhat PATH cua process hien tai
        $env:Path = [System.Environment]::GetEnvironmentVariable("Path", "Machine") + ";" + `
                    [System.Environment]::GetEnvironmentVariable("Path", "User")

    } catch {
        Write-Host "    Loi khi dat JAVA_HOME: $_" -ForegroundColor Red
    }
}

function Test-InstalledTools {
    <# Kiem tra phien ban cac cong cu da cai #>
    Write-Step "Kiem tra phien ban cac cong cu..." "INFO"

    # Lam moi PATH
    $env:Path = [System.Environment]::GetEnvironmentVariable("Path", "Machine") + ";" + `
                [System.Environment]::GetEnvironmentVariable("Path", "User")

    Write-Host ""
    Write-Host "    === KIEM TRA CONG CU ===" -ForegroundColor Cyan

    # Git
    if (Test-CommandExists "git") {
        $gitVer = git --version 2>$null
        Write-Host "    git:    $gitVer" -ForegroundColor Green
    } else {
        Write-Host "    git:    KHONG TIM THAY" -ForegroundColor Yellow
    }

    # Docker
    if (Test-CommandExists "docker") {
        $dockerVer = docker --version 2>$null
        Write-Host "    docker: $dockerVer" -ForegroundColor Green
    } else {
        Write-Host "    docker: KHONG TIM THAY (co the can khoi dong lai)" -ForegroundColor Yellow
    }

    # Java
    if (Test-CommandExists "java") {
        $javaVer = java -version 2>&1 | Select-Object -First 1
        Write-Host "    java:   $javaVer" -ForegroundColor Green
    } else {
        Write-Host "    java:   KHONG TIM THAY" -ForegroundColor Yellow
    }

    Write-Host ""
}

function Show-Summary {
    <# In bang tong ket 4 nhom ket qua #>
    Write-Host ""
    Write-Host "================================================================" -ForegroundColor Cyan
    Write-Host "                     BANG TONG KET SETUP                        " -ForegroundColor Cyan
    Write-Host "================================================================" -ForegroundColor Cyan

    Write-Host ""
    Write-Host "[DA CAI MOI]" -ForegroundColor Green
    if ($script:results.Installed.Count -gt 0) {
        foreach ($item in $script:results.Installed) {
            Write-Host "  + $item" -ForegroundColor Green
        }
    } else {
        Write-Host "  (khong co)" -ForegroundColor Gray
    }

    Write-Host ""
    Write-Host "[DA CO SAN - BO QUA]" -ForegroundColor Yellow
    if ($script:results.AlreadyHave.Count -gt 0) {
        foreach ($item in $script:results.AlreadyHave) {
            Write-Host "  ~ $item" -ForegroundColor Yellow
        }
    } else {
        Write-Host "  (khong co)" -ForegroundColor Gray
    }

    Write-Host ""
    Write-Host "[THAT BAI]" -ForegroundColor Red
    if ($script:results.Failed.Count -gt 0) {
        foreach ($item in $script:results.Failed) {
            Write-Host "  X $item" -ForegroundColor Red
        }
    } else {
        Write-Host "  (khong co)" -ForegroundColor Gray
    }

    Write-Host ""
    Write-Host "[CAN CAI THU CONG]" -ForegroundColor Magenta
    if ($script:results.ManualNeeded.Count -gt 0) {
        foreach ($item in $script:results.ManualNeeded) {
            Write-Host "  ! $item" -ForegroundColor Magenta
        }
    } else {
        Write-Host "  (khong co)" -ForegroundColor Gray
    }

    Write-Host ""
    Write-Host "================================================================" -ForegroundColor Cyan

    if ($script:needReboot) {
        Write-Host ""
        Write-Host "[!!!] CAN KHOI DONG LAI MAY de hoan tat (WSL / VirtualMachinePlatform)." -ForegroundColor Yellow
        Write-Host "      Sau khi khoi dong lai, mo Docker Desktop de hoan tat cai dat." -ForegroundColor Yellow
    }

    Write-Host ""
    Write-Host "Log file: $logFile" -ForegroundColor Cyan
    Write-Host ""
}
