    [Security.Principal.WindowsIdentity]::GetCurrent()
)

    [Security.Principal.WindowsBuiltInRole]::Administrator)) {

    Write-Host "   Hay click phai PowerShell -> Run as Administrator" -ForegroundColor Yellow

    exit 1

# Biến theo dõi kết quả
$script:Results = @()
$script:Errors  = @()

function Log-Step {
    param([string]$Step, [string]$Status, [string]$Detail)
    $entry = [PSCustomObject]@{
        Buoc      = $Step
        TrangThai = $Status
        ChiTiet   = $Detail
    }
    $script:Results += $entry
    $color = if ($Status -eq "OK") { "Green" } 
             elseif ($Status -eq "SKIP") { "Yellow" } 
             else { "Red" }
    Write-Host "[$Status] $Step — $Detail" -ForegroundColor $color
}

function Log-Error {
    param([string]$Step, [string]$ErrorMsg)
    $script:Errors += [PSCustomObject]@{ Buoc = $Step; Loi = $ErrorMsg }
    Write-Host "[LOI] $Step — $ErrorMsg" -ForegroundColor Red
}

# ============================================================================
Write-Host "=" * 70 -ForegroundColor DarkGray
Write-Host "BUOC 0: Xac minh o dia he thong" -ForegroundColor Cyan
Write-Host "=" * 70 -ForegroundColor DarkGray

$diskInfo = Get-PhysicalDisk | Select-Object DeviceId, MediaType, BusType, Size
$diskInfo | Format-Table -AutoSize

# Tìm ổ chứa C:
$systemDiskNumber = (Get-Partition -DriveLetter C -ErrorAction SilentlyContinue).DiskNumber
$systemDisk = Get-PhysicalDisk | Where-Object { $_.DeviceId -eq $systemDiskNumber }

$isHDD = $false
if ($null -eq $systemDisk) {
    Write-Host "⚠ Khong xac dinh duoc loai o dia he thong. Gia su la HDD." -ForegroundColor Yellow
    $isHDD = $true
} elseif ($systemDisk.MediaType -eq "SSD") {
    Write-Host "⛔ O dia he thong la SSD!" -ForegroundColor Red
    Write-Host "   Cac toi uu trong script nay danh cho HDD." -ForegroundColor Red
    Write-Host "   Toi uu cho SSD nguoc lai hoan toan (bat TRIM, tat defrag, tat Superfetch)." -ForegroundColor Red
    $confirm = Read-Host "Ban co muon tiep tuc voi cac toi uu CHUNG (bo qua phan rieng HDD)? (Y/N)"
    if ($confirm -ne "Y" -and $confirm -ne "y") {
        Write-Host "Da dung lai theo yeu cau." -ForegroundColor Yellow
        exit 0
    }
    $isHDD = $false
} else {
    # MediaType = "HDD" hoặc "Unspecified"
    Write-Host "✓ O dia he thong: $($systemDisk.MediaType) — Tiep tuc toi uu cho HDD" -ForegroundColor Green
    $isHDD = $true
}

# ============================================================================
# BƯỚC 1: TẠO SYSTEM RESTORE POINT
# ============================================================================
Write-Host ""
Write-Host "=" * 70 -ForegroundColor DarkGray
Write-Host "BUOC 1: Tao System Restore Point" -ForegroundColor Cyan
Write-Host "=" * 70 -ForegroundColor DarkGray

try {
    # Bật System Restore trên ổ C: nếu chưa bật
    Enable-ComputerRestore -Drive "C:\" -ErrorAction Stop
    Write-Host "  > Enable-ComputerRestore -Drive 'C:\'" -ForegroundColor DarkYellow

    # Tạo Restore Point
    # Lưu ý: Windows giới hạn 1 restore point / 24h nếu đã có, nên dùng 
    # registry trick để bypass
    $rpFreqPath = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore"
    $origFreq = (Get-ItemProperty -Path $rpFreqPath -Name "SystemRestorePointCreationFrequency" -ErrorAction SilentlyContinue).SystemRestorePointCreationFrequency
    Set-ItemProperty -Path $rpFreqPath -Name "SystemRestorePointCreationFrequency" -Value 0 -Type DWord -Force

    Checkpoint-Computer -Description "Pre-Optimization_$(Get-Date -Format 'yyyyMMdd_HHmmss')" -RestorePointType "MODIFY_SETTINGS" -ErrorAction Stop
    Write-Host "  > Checkpoint-Computer -Description 'Pre-Optimization'" -ForegroundColor DarkYellow

    # Khôi phục giá trị gốc
    if ($null -ne $origFreq) {
        Set-ItemProperty -Path $rpFreqPath -Name "SystemRestorePointCreationFrequency" -Value $origFreq -Type DWord -Force
    } else {
        Remove-ItemProperty -Path $rpFreqPath -Name "SystemRestorePointCreationFrequency" -ErrorAction SilentlyContinue
    }

    Log-Step "1. Restore Point" "OK" "Da tao thanh cong"
} catch {
    Log-Error "1. Restore Point" $_.Exception.Message
    Log-Step "1. Restore Point" "LOI" "Khong tao duoc Restore Point — $($_.Exception.Message)"
    Write-Host "⚠ Tiep tuc cac buoc con lai..." -ForegroundColor Yellow
}

# ============================================================================
# BƯỚC 2: CÀI ĐẶT SCOOP + THÊM BUCKETS
# ============================================================================
Write-Host ""
Write-Host "=" * 70 -ForegroundColor DarkGray
Write-Host "BUOC 2: Cai dat Scoop va them buckets" -ForegroundColor Cyan
Write-Host "=" * 70 -ForegroundColor DarkGray

try {
    $scoopCmd = Get-Command scoop -ErrorAction SilentlyContinue
    if ($null -eq $scoopCmd) {
        Write-Host "  Scoop chua duoc cai. Dang cai dat..." -ForegroundColor Yellow
        Write-Host "  > Invoke-RestMethod get.scoop.sh | Invoke-Expression" -ForegroundColor DarkYellow
        
        # Scoop cần chạy dưới user thường (non-admin). Nếu đang chạy admin, 
        # cần thêm tham số
        $env:SCOOP = "$env:USERPROFILE\scoop"
        [Environment]::SetEnvironmentVariable("SCOOP", $env:SCOOP, "User")
        
        Invoke-RestMethod -Uri "https://get.scoop.sh" -OutFile "$env:TEMP\install-scoop.ps1"
        & powershell.exe -ExecutionPolicy Bypass -File "$env:TEMP\install-scoop.ps1" -RunAsAdmin
        
        # Refresh PATH
        $env:Path = [System.Environment]::GetEnvironmentVariable("Path","Machine") + ";" + [System.Environment]::GetEnvironmentVariable("Path","User")
        
        Log-Step "2a. Cai Scoop" "OK" "Da cai dat Scoop"
    } else {
        Write-Host "  ✓ Scoop da duoc cai san tai: $($scoopCmd.Source)" -ForegroundColor Green
        Log-Step "2a. Cai Scoop" "OK" "Da co san"
    }

    # Thêm bucket extras và versions
    Write-Host "  > scoop bucket add extras" -ForegroundColor DarkYellow
    & scoop bucket add extras 2>&1 | Out-Null
    Write-Host "  > scoop bucket add versions" -ForegroundColor DarkYellow
    & scoop bucket add versions 2>&1 | Out-Null
    
    Log-Step "2b. Them buckets" "OK" "Da them extras, versions"
} catch {
    Log-Error "2. Scoop" $_.Exception.Message
    Log-Step "2. Scoop" "LOI" $_.Exception.Message
}

# BƯỚC 3: CÀI ỨNG DỤNG QUA SCOOP
# ============================================================================
Write-Host ""
Write-Host "=" * 70 -ForegroundColor DarkGray
Write-Host "BUOC 3: Cai ung dung qua Scoop" -ForegroundColor Cyan
Write-Host "=" * 70 -ForegroundColor DarkGray

        Write-Host "  > scoop install $app" -ForegroundColor DarkYellow

        $outputStr = $output -join "`n"
        
        if ($outputStr -match "already installed" -or $outputStr -match "was installed successfully") {
            Log-Step "3. Scoop: $app" "OK" "Da cai/da co san"
        } else {
            # Kiểm tra xem có lỗi không
            if ($LASTEXITCODE -ne 0 -and $null -ne $LASTEXITCODE) {
                Log-Step "3. Scoop: $app" "LOI" "Exit code: $LASTEXITCODE"
            } else {
                Log-Step "3. Scoop: $app" "OK" "Da cai dat"
            }
        }
    } catch {
        Log-Error "3. Scoop: $app" $_.Exception.Message
        Log-Step "3. Scoop: $app" "LOI" $_.Exception.Message

    Set-ItemProperty -Path "HKCU:\Control Panel\Desktop" -Name "SmoothScroll" -Value 0 -Type DWord -Force -ErrorAction SilentlyContinue
    
    # Tắt animation window
    $sysMetricsPath = "HKCU:\Control Panel\Desktop\WindowMetrics"
    Set-ItemProperty -Path $sysMetricsPath -Name "MinAnimate" -Value "0" -Type String -Force

    Log-Step "5b. Visual Effects" "OK" "Da chuyen sang Best Performance, tat animations"

# ---- 5h. GIỮ NGUYÊN SysMain/Superfetch + Prefetch (QUAN TRỌNG cho HDD) ----
Write-Host ""
Write-Host "  [5h] Gia co SysMain/Superfetch + Prefetch (GIU BAT — dung cho HDD)" -ForegroundColor White
if ($isHDD) {
    try {
        # Đảm bảo SysMain đang chạy
        $sysMainSvc = Get-Service -Name "SysMain" -ErrorAction SilentlyContinue
        if ($null -eq $sysMainSvc) {
            # Trên Windows 10 cũ, service tên là Superfetch
            $sysMainSvc = Get-Service -Name "Superfetch" -ErrorAction SilentlyContinue
        }
        
        if ($null -ne $sysMainSvc) {
            Write-Host "  > Set-Service $($sysMainSvc.Name) -StartupType Automatic" -ForegroundColor DarkYellow
            Set-Service $sysMainSvc.Name -StartupType Automatic -ErrorAction Stop
            if ($sysMainSvc.Status -ne "Running") {
                Start-Service $sysMainSvc.Name -ErrorAction SilentlyContinue
            }
        }
        
        # Đảm bảo Prefetch bật ở mức tối đa
        $prefetchPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management\PrefetchParameters"
        Write-Host "  > Set EnablePrefetcher = 3, EnableSuperfetch = 3" -ForegroundColor DarkYellow
        Set-ItemProperty -Path $prefetchPath -Name "EnablePrefetcher" -Value 3 -Type DWord -Force
        Set-ItemProperty -Path $prefetchPath -Name "EnableSuperfetch" -Value 3 -Type DWord -Force

        Log-Step "5h. SysMain/Prefetch" "OK" "GIU BAT — SysMain=Automatic, Prefetcher=3, Superfetch=3"
    } catch {
        Log-Error "5h. SysMain/Prefetch" $_.Exception.Message
        Log-Step "5h. SysMain/Prefetch" "LOI" $_.Exception.Message
    }
} else {
    Write-Host "  (Khong phai HDD — bo qua buoc gia co SysMain)" -ForegroundColor DarkGray
    Log-Step "5h. SysMain/Prefetch" "SKIP" "Khong phai HDD"
}

    Set-ItemProperty -Path $lockScreenPath -Name "NoLockScreen" -Value 1 -Type DWord -Force -ErrorAction SilentlyContinue

    $peoplePath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced\People"
    if (-not (Test-Path $peoplePath)) { New-Item -Path $peoplePath -Force | Out-Null }
    Set-ItemProperty -Path $peoplePath -Name "PeopleBand" -Value 0 -Type DWord -Force

    # Tắt News and Interests trên taskbar (Windows 10 21H1+)
    $feedsPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Feeds"
    if (-not (Test-Path $feedsPath)) { New-Item -Path $feedsPath -Force | Out-Null }
    Set-ItemProperty -Path $feedsPath -Name "ShellFeedsTaskbarViewMode" -Value 2 -Type DWord -Force

    # Tắt Startup delay
    $serializePath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\Serialize"
    if (-not (Test-Path $serializePath)) { New-Item -Path $serializePath -Force | Out-Null }
    Set-ItemProperty -Path $serializePath -Name "StartupDelayInMSec" -Value 0 -Type DWord -Force

    Log-Step "5j. Toi uu bo sung" "OK" "Tat tips, suggestions, news, People, startup delay"
} catch {
    Log-Error "5j. Toi uu bo sung" $_.Exception.Message
    Log-Step "5j. Toi uu bo sung" "LOI" $_.Exception.Message
}

# ---- 5k. Tối ưu Chrome cho HDD (giảm I/O đĩa) ----
Write-Host ""
Write-Host "  [5k] Toi uu Google Chrome cho HDD (giam I/O dia)" -ForegroundColor White
try {
    # --- 5k-1. Tắt Software Reporter Tool (chrome_cleanup_tool) ---
    # Tool này quét toàn bộ ổ đĩa, cực kỳ nặng I/O trên HDD
    $srtPath = "HKLM:\SOFTWARE\Policies\Google\Chrome"
    if (-not (Test-Path $srtPath)) { New-Item -Path $srtPath -Force | Out-Null }
    Write-Host "  > Tat Software Reporter Tool (ChromeCleanupEnabled=0)" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "ChromeCleanupEnabled" -Value 0 -Type DWord -Force
    Set-ItemProperty -Path $srtPath -Name "ChromeCleanupReportingEnabled" -Value 0 -Type DWord -Force

    # Chặn file exe của Software Reporter Tool
    $srtExeDir = "$env:LOCALAPPDATA\Google\Chrome\User Data\SwReporter"
    if (Test-Path $srtExeDir) {
        Write-Host "  > Chan quyen thuc thi SwReporter" -ForegroundColor DarkYellow
        icacls $srtExeDir /deny "*S-1-1-0:(OI)(CI)(RX)" 2>&1 | Out-Null
    }

    # --- 5k-2. Giới hạn Disk Cache xuống 100MB ---
    # Mặc định Chrome cache không giới hạn, gây ghi liên tục trên HDD
    Write-Host "  > Gioi han Disk Cache = 104857600 (100MB)" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "DiskCacheSize" -Value 104857600 -Type DWord -Force

    # --- 5k-3. Tắt Prefetch / Prerender trang web ---
    # Giảm I/O bằng cách không tải trước các trang
    Write-Host "  > Tat NetworkPrediction (prefetch/prerender)" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "NetworkPredictionOptions" -Value 2 -Type DWord -Force
    # 0=Always, 1=Wifi only, 2=Never

    # --- 5k-4. Tắt Chrome chạy nền khi đóng trình duyệt ---
    Write-Host "  > Tat BackgroundModeEnabled" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "BackgroundModeEnabled" -Value 0 -Type DWord -Force

    # --- 5k-5. Tắt tự động cập nhật component (giảm ghi đĩa nền) ---
    Write-Host "  > Tat ComponentUpdatesEnabled" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "ComponentUpdatesEnabled" -Value 0 -Type DWord -Force

    # --- 5k-6. Tắt đồng bộ tự động (giảm ghi đĩa liên tục) ---
    # Người dùng vẫn có thể bật lại thủ công trong Chrome
    Write-Host "  > Tat SyncDisabled (giam ghi dia lien tuc)" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "SyncDisabled" -Value 1 -Type DWord -Force

    # --- 5k-7. Tắt metrics/reporting (giảm ghi log nền) ---
    Write-Host "  > Tat MetricsReportingEnabled" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "MetricsReportingEnabled" -Value 0 -Type DWord -Force

    # --- 5k-8. Giảm số process renderer (giảm RAM + swap trên HDD) ---
    Write-Host "  > Gioi han RendererProcessLimit = 4" -ForegroundColor DarkYellow
    Set-ItemProperty -Path $srtPath -Name "RendererProcessLimit" -Value 4 -Type DWord -Force

    # --- 5k-9. Thêm launch flags tối ưu vào Chrome shortcut ---
    # Tạo file batch để user có thể dùng
    $chromeFlags = @(
        "--disk-cache-size=104857600",   # 100MB cache
        "--disable-background-networking",
        "--disable-component-update",
        "--disable-features=OptimizationHints,MediaRouter"
    )
    $chromeFlagsStr = $chromeFlags -join " "

    # Tìm Chrome exe
    $chromeExe = $null
    $chromePaths = @(
        "$env:ProgramFiles\Google\Chrome\Application\chrome.exe",
        "${env:ProgramFiles(x86)}\Google\Chrome\Application\chrome.exe",
        "$env:LOCALAPPDATA\Google\Chrome\Application\chrome.exe"
    )
    foreach ($p in $chromePaths) {
        if (Test-Path $p) { $chromeExe = $p; break }
    }

    if ($null -ne $chromeExe) {
        # Tạo shortcut tối ưu trên Desktop
        $desktopPath = [Environment]::GetFolderPath("Desktop")
        $shortcutPath = "$desktopPath\Chrome (HDD Optimized).lnk"
        $WshShell = New-Object -ComObject WScript.Shell
        $shortcut = $WshShell.CreateShortcut($shortcutPath)
        $shortcut.TargetPath = $chromeExe
        $shortcut.Arguments = $chromeFlagsStr
        $shortcut.Description = "Google Chrome - Toi uu cho HDD"
        $shortcut.Save()
        Write-Host "  > Da tao shortcut: $shortcutPath" -ForegroundColor DarkYellow
    }

    Log-Step "5k. Chrome HDD Optimize" "OK" "Tat SRT, cache 100MB, tat prefetch/background/sync/metrics, gioi han renderer"
} catch {
    Log-Error "5k. Chrome HDD Optimize" $_.Exception.Message
    Log-Step "5k. Chrome HDD Optimize" "LOI" $_.Exception.Message
}

# 6a. %TEMP%
try {
    $tempPath = $env:TEMP
    Write-Host "  > Don $tempPath" -ForegroundColor DarkYellow
    $tempFiles = Get-ChildItem -Path $tempPath -Recurse -Force -ErrorAction SilentlyContinue
    $tempSize = ($tempFiles | Measure-Object -Property Length -Sum -ErrorAction SilentlyContinue).Sum
    $tempFiles | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
    $cleanedMB = [math]::Round($tempSize / 1MB, 2)
    $totalCleaned += $tempSize
    Log-Step "6a. %TEMP%" "OK" "Da don ~${cleanedMB} MB"
